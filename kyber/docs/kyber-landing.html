MINIMAL AGENTPYTHON
from kyber.core.vsm import (
    ViableAgent, AgentIdentity,
    VSMLevel, Task
)

class MyAgent(ViableAgent):
    def __init__(self):
        super().__init__(AgentIdentity(
            name="MY_AGENT",
            vsm_level=VSMLevel.S1_OPERATIONS,
            vsm_depth=1,
            capabilities=["analyze"],
            charter="Do excellent work.",
        ))

    async def s1_execute(self, task: Task):
        return f"Done: {task.intent}"
